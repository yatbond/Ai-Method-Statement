/*
  Implementation note — updated 2026-05-04

  This file remains the frontend redesign prototype/reference. The running app now
  implements the core navigation model from this prototype:
  - Dashboard
  - Projects
  - Trades
  - Draft Editor
  - Conflict Review
  - Export Review
  - Settings

  Current implementation additions since this prototype:
  - Settings includes Import, Document AI Settings, AI Services, File Storage
    test, ingestion status, batch filters, terminate controls, and hosted-safe
    browser PDF upload.
  - Draft Editor supports section-level user input before drafting. The LLM uses
    user input, uploaded project documents, selected precedent passages, gaps,
    and conflicts to draft sections with explicit [GAP:] markers.
  - Trades / Knowledge Base document pages paginate passages instead of showing
    only the first 20 passages.
  - Conflict Review is kept as a separate review surface after drafting.
  - Browser upload is the preferred hosted workflow. Local folder import remains
    a local-development convenience only.
*/

function Badge({ children, tone = "slate" }) {
  const classes = {
    slate: "bg-slate-100 text-slate-700 border-slate-200",
    green: "bg-emerald-50 text-emerald-700 border-emerald-200",
    amber: "bg-amber-50 text-amber-700 border-amber-200",
    red: "bg-red-50 text-red-700 border-red-200",
    blue: "bg-blue-50 text-blue-700 border-blue-200",
    purple: "bg-purple-50 text-purple-700 border-purple-200",
  };
  return <span className={`inline-flex items-center rounded-full border px-2.5 py-1 text-xs font-medium ${classes[tone] || classes.slate}`}>{children}</span>;
}

function Card({ children, className = "" }) {

}
